HOST_ACME_DOMAIN_CERT_MODULUS $(openssl x509 -in fullchain.cer -noout -modulus)
HOST_ACME_DOMAIN_KEY_MODULUS $(openssl rsa -in ${domain}.key -noout -modulus)
HOST_ACME_DOMAIN_PATH_VALID $(echo ${DOMAIN_PATH:-} |awk "'"/^[0-9a-z_\-\.+\/]+@[0-9a-z_\-\.]+\.[a-z0-9_\-\.\+\/]+$/"'")
HOST_ACME_POST_HOOK [ "$(echo ${DOMAIN_PATH:-} |awk "'"/^[0-9a-z_\-\.+\/]+@[0-9a-z_\-\.]+\.[a-z0-9_\-\.\+\/]+$/"'")" ] && cp -a fullchain.cer /host/certs/${domain}-cert.pem 2>/dev/null && [ "$(openssl x509 -in fullchain.cer -noout -modulus)" = "$(openssl rsa -in ${domain}.key -noout -modulus)" ] && cp -a ${domain}.key /host/certs/${domain}-key.pem
[exit 0]
